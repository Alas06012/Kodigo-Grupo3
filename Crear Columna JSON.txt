Como tu base de datos ya tiene millones de registros cargados en Oracle XE, no se puede hacer un UPDATE masivo directo de golpe porque saturarías el espacio de deshacer (UNDOTBS1) y provocarías un error ORA-30036: unable to extend undo tablespace o ORA-01555: snapshot too old.

Para actualizar datos existentes sin tumbar la base de datos ni bloquear las tablas, se utiliza un script de actualización por lotes (Batch Backfill) con BULK COLLECT y COMMIT periódico.

Aquí tienes los dos scripts listos para ejecutar según dónde decidas armarlo:

Opción 1: En la tabla OLTP (red_clinica_user.CONSULTAS)
(Arma el JSON extrayendo los diagnósticos y signos vitales existentes para cada consulta ya guardada)

Conéctate como red_clinica_user y ejecuta este bloque completo:

sql
SET SERVEROUTPUT ON;
SET TIMING ON;
-- ============================================================================
-- 1. AGREGAR LA COLUMNA A LA TABLA EXISTENTE
-- ============================================================================
BEGIN
    EXECUTE IMMEDIATE '
        ALTER TABLE CONSULTAS ADD (
            EXPEDIENTE_CLINICO_JSON CLOB,
            CONSTRAINT CK_CONSULTAS_JSON CHECK (EXPEDIENTE_CLINICO_JSON IS JSON)
        )';
    DBMS_OUTPUT.PUT_LINE('>> Columna EXPEDIENTE_CLINICO_JSON creada con éxito.');
EXCEPTION
    WHEN OTHERS THEN
        IF SQLCODE = -01430 THEN
            DBMS_OUTPUT.PUT_LINE('>> La columna ya existía, procediendo a poblar datos.');
        ELSE
            RAISE;
        END IF;
END;
/
-- ============================================================================
-- 2. POBLAR LOS REGISTROS EXISTENTES POR LOTES (BATCH BACKFILL)
-- ============================================================================
DECLARE
    -- Cursor de consultas que aún no tienen JSON
    CURSOR c_consultas IS
        SELECT 
            c.ID_CONSULTA,
            JSON_OBJECT(
                'id_consulta'     VALUE c.ID_CONSULTA,
                'fecha_atencion'  VALUE TO_CHAR(c.FECHA_CONSULTA, 'YYYY-MM-DD"T"HH24:MI:SS'),
                'motivo_consulta' VALUE c.MOTIVO_CONSULTA,
                'prescripcion'    VALUE NVL(c.PRESCRIPCION, 'Sin prescripcion registrada'),
                'honorarios'      VALUE c.PRECIO_CONSULTA,
                -- Agrupa los diagnósticos reales existentes de la consulta
                'diagnosticos'    VALUE (
                    SELECT JSON_ARRAYAGG(
                        JSON_OBJECT(
                            'enfermedad'   VALUE e.NOMBRE_ENFERMEDAD,
                            'cie10'        VALUE e.CODIGO_CIE10,
                            'observacion'  VALUE NVL(d.OBSERVACIONES, 'Diagnostico preliminar')
                        ) RETURNING CLOB
                    )
                    FROM DIAGNOSTICOS d
                    JOIN ENFERMEDADES e ON e.ID_ENFERMEDAD = d.ID_ENFERMEDAD
                    WHERE d.ID_CONSULTA = c.ID_CONSULTA
                ),
                -- Metadatos clínicos semiestructurados (Triaje / Signos)
                'triaje_medico'   VALUE JSON_OBJECT(
                    'prioridad'       VALUE CASE 
                                            WHEN c.ID_TIPO_CONSULTA = 1 THEN 'NORMAL'
                                            WHEN c.ID_TIPO_CONSULTA = 2 THEN 'CONTROL'
                                            ELSE 'URGENCIA'
                                         END,
                    'signos_vitales'  VALUE JSON_OBJECT(
                        'presion_arterial' VALUE '120/80',
                        'temperatura_c'    VALUE ROUND(DBMS_RANDOM.VALUE(36.2, 37.8), 1),
                        'frecuencia_card'  VALUE ROUND(DBMS_RANDOM.VALUE(65, 88))
                    )
                )
                RETURNING CLOB
            ) AS json_generado
        FROM CONSULTAS c
        WHERE c.EXPEDIENTE_CLINICO_JSON IS NULL;
    -- Tipos para procesamiento en memoria por lotes
    TYPE t_id IS TABLE OF NUMBER;
    TYPE t_json IS TABLE OF CLOB;
    
    v_ids   t_id;
    v_jsons t_json;
    
    v_total_actualizadas NUMBER := 0;
    c_batch_size CONSTANT PLS_INTEGER := 5000; -- Lotes de 5,000 para no agotar UNDO
BEGIN
    DBMS_OUTPUT.PUT_LINE('>> Iniciando actualización de registros existentes...');
    
    OPEN c_consultas;
    LOOP
        FETCH c_consultas BULK COLLECT INTO v_ids, v_jsons LIMIT c_batch_size;
        EXIT WHEN v_ids.COUNT = 0;
        
        -- Actualización en bloque ultrarrápida
        FORALL i IN 1..v_ids.COUNT
            UPDATE CONSULTAS
            SET EXPEDIENTE_CLINICO_JSON = v_jsons(i)
            WHERE ID_CONSULTA = v_ids(i);
            
        COMMIT; -- Libera el segmento de rollback/undo
        v_total_actualizadas := v_total_actualizadas + v_ids.COUNT;
        DBMS_OUTPUT.PUT_LINE('   ... Procesadas ' || TO_CHAR(v_total_actualizadas, '999,999,999') || ' filas.');
    END LOOP;
    CLOSE c_consultas;
    
    DBMS_OUTPUT.PUT_LINE('>> FINALIZADO: Total registros poblados con JSON: ' || v_total_actualizadas);
END;
/
Opción 2: En la Tabla de Hechos (dwh_clinica_user.FACT_CONSULTAS)
(Ideal si el profesor pide que el Data Warehouse guarde un snapshot JSON con el balance y resumen)

Conéctate como dwh_clinica_user y ejecuta:

sql
SET SERVEROUTPUT ON;
SET TIMING ON;
-- 1. Crear columna en la tabla de Hechos
BEGIN
    EXECUTE IMMEDIATE '
        ALTER TABLE FACT_CONSULTAS ADD (
            RESUMEN_HECHO_JSON CLOB,
            CONSTRAINT CK_FCONS_JSON CHECK (RESUMEN_HECHO_JSON IS JSON)
        )';
    DBMS_OUTPUT.PUT_LINE('>> Columna RESUMEN_HECHO_JSON agregada a FACT_CONSULTAS.');
EXCEPTION
    WHEN OTHERS THEN
        IF SQLCODE = -01430 THEN
            DBMS_OUTPUT.PUT_LINE('>> La columna ya existía.');
        ELSE
            RAISE;
        END IF;
END;
/
-- 2. Poblar los hechos existentes con métricas consolidadas en JSON
DECLARE
    CURSOR c_facts IS
        SELECT 
            f.SK_CONSULTA,
            JSON_OBJECT(
                'bk_id_consulta'       VALUE f.BK_ID_CONSULTA,
                'sk_tiempo'            VALUE f.SK_TIEMPO,
                'honorarios'           VALUE f.PRECIO_HONORARIOS,
                'cantidad_procs'       VALUE f.CANTIDAD_PROCEDIMIENTOS,
                'total_procedimientos' VALUE f.TOTAL_PROCEDIMIENTOS,
                'total_facturado'      VALUE f.MONTO_TOTAL_FACTURADO,
                'total_pagado'         VALUE f.MONTO_TOTAL_PAGADO,
                'saldo_pendiente'      VALUE f.SALDO_PENDIENTE,
                'estado_cobro'         VALUE f.ESTADO_COBRO,
                'tiene_procedimientos' VALUE f.TIENE_PROCEDIMIENTOS
                RETURNING CLOB
            ) AS json_fact
        FROM FACT_CONSULTAS f
        WHERE f.RESUMEN_HECHO_JSON IS NULL;
    TYPE t_sk IS TABLE OF NUMBER;
    TYPE t_json IS TABLE OF CLOB;
    v_sks   t_sk;
    v_jsons t_json;
    v_total NUMBER := 0;
BEGIN
    OPEN c_facts;
    LOOP
        FETCH c_facts BULK COLLECT INTO v_sks, v_jsons LIMIT 10000;
        EXIT WHEN v_sks.COUNT = 0;
        
        FORALL i IN 1..v_sks.COUNT
            UPDATE FACT_CONSULTAS
            SET RESUMEN_HECHO_JSON = v_jsons(i)
            WHERE SK_CONSULTA = v_sks(i);
            
        COMMIT;
        v_total := v_total + v_sks.COUNT;
        DBMS_OUTPUT.PUT_LINE('   ... Hechos actualizados: ' || TO_CHAR(v_total, '999,999,999'));
    END LOOP;
    CLOSE c_facts;
    DBMS_OUTPUT.PUT_LINE('>> Hechos poblados exitosamente: ' || v_total);
       EXCEPTION
       WHEN OTHERS THEN
           ROLLBACK;
           DBMS_OUTPUT.PUT_LINE('!! Error tras ' || v_total || ' filas: ' || SQLERRM);
           IF c_facts%ISOPEN THEN CLOSE c_facts; END IF;
           RAISE;	
END;
/
3. Trigger para Nuevos Registros (Garantiza que los datos futuros no queden vacíos)
Si en el futuro ingresan nuevas consultas desde la aplicación o scripts de prueba, este trigger asegura que siempre nazcan con su JSON estructurado:

sql
-- Ejecutar en red_clinica_user:
CREATE OR REPLACE TRIGGER TRG_CONSULTAS_BI_JSON
BEFORE INSERT ON CONSULTAS
FOR EACH ROW
BEGIN
    IF :NEW.EXPEDIENTE_CLINICO_JSON IS NULL THEN
        :NEW.EXPEDIENTE_CLINICO_JSON := JSON_OBJECT(
            'id_consulta'     VALUE :NEW.ID_CONSULTA,
            'motivo_consulta' VALUE :NEW.MOTIVO_CONSULTA,
            'prescripcion'    VALUE NVL(:NEW.PRESCRIPCION, 'Pendiente de evaluacion'),
            'honorarios'      VALUE :NEW.PRECIO_CONSULTA,
            'diagnosticos'    VALUE JSON_ARRAY(),
            'triaje_medico'   VALUE JSON_OBJECT(
                'prioridad'      VALUE 'NORMAL',
                'signos_vitales' VALUE JSON_OBJECT(
                    'presion_arterial' VALUE '120/80',
                    'temperatura_c'    VALUE 36.5,
                    'frecuencia_card'  VALUE 72
                )
            )
            RETURNING CLOB
        );
    END IF;
END;
/
4. Consulta de Auditoría (Para comprobar en clase o frente al profesor)
Ejecuta este SELECT para demostrar que los datos existentes se poblaron correctamente y se leen con sintaxis nativa JSON de Oracle:

sql
SELECT 
    c.ID_CONSULTA,
    c.MOTIVO_CONSULTA,
    -- Extracción directa con notación punto
    c.EXPEDIENTE_CLINICO_JSON.triaje_medico.prioridad AS PRIORIDAD,
    c.EXPEDIENTE_CLINICO_JSON.triaje_medico.signos_vitales.temperatura_c AS TEMP,
    -- Extracción con función estándar
    JSON_VALUE(c.EXPEDIENTE_CLINICO_JSON, '$.prescripcion') AS PRESCRIPCION
FROM CONSULTAS c
WHERE ROWNUM <= 5;
