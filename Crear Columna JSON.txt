Opción 1: En la Tabla de Hechos (dwh_clinica_user.FACT_CONSULTAS)
(Ideal si el profesor pide que el Data Warehouse guarde un snapshot JSON con el balance y resumen)

Conéctate como dwh_clinica_user y ejecuta:

sql
SET SERVEROUTPUT ON;
SET TIMING ON;
-- 1. Crear columna en la tabla de Hechos
BEGIN
    EXECUTE IMMEDIATE '
        ALTER TABLE FACT_CONSULTAS ADD (
            RESUMEN_HECHO_JSON CLOB,
            CONSTRAINT CK_FCONS_JSON CHECK (RESUMEN_HECHO_JSON IS JSON)
        )';
    DBMS_OUTPUT.PUT_LINE('>> Columna RESUMEN_HECHO_JSON agregada a FACT_CONSULTAS.');
EXCEPTION
    WHEN OTHERS THEN
        IF SQLCODE = -01430 THEN
            DBMS_OUTPUT.PUT_LINE('>> La columna ya existía.');
        ELSE
            RAISE;
        END IF;
END;
/
-- 2. Poblar los hechos existentes con métricas consolidadas en JSON
DECLARE
    CURSOR c_facts IS
        SELECT 
            f.SK_CONSULTA,
            JSON_OBJECT(
                'bk_id_consulta'       VALUE f.BK_ID_CONSULTA,
                'sk_tiempo'            VALUE f.SK_TIEMPO,
                'honorarios'           VALUE f.PRECIO_HONORARIOS,
                'cantidad_procs'       VALUE f.CANTIDAD_PROCEDIMIENTOS,
                'total_procedimientos' VALUE f.TOTAL_PROCEDIMIENTOS,
                'total_facturado'      VALUE f.MONTO_TOTAL_FACTURADO,
                'total_pagado'         VALUE f.MONTO_TOTAL_PAGADO,
                'saldo_pendiente'      VALUE f.SALDO_PENDIENTE,
                'estado_cobro'         VALUE f.ESTADO_COBRO,
                'tiene_procedimientos' VALUE f.TIENE_PROCEDIMIENTOS
                RETURNING CLOB
            ) AS json_fact
        FROM FACT_CONSULTAS f
        WHERE f.RESUMEN_HECHO_JSON IS NULL;
    TYPE t_sk IS TABLE OF NUMBER;
    TYPE t_json IS TABLE OF CLOB;
    v_sks   t_sk;
    v_jsons t_json;
    v_total NUMBER := 0;
BEGIN
    OPEN c_facts;
    LOOP
        FETCH c_facts BULK COLLECT INTO v_sks, v_jsons LIMIT 10000;
        EXIT WHEN v_sks.COUNT = 0;
        
        FORALL i IN 1..v_sks.COUNT
            UPDATE FACT_CONSULTAS
            SET RESUMEN_HECHO_JSON = v_jsons(i)
            WHERE SK_CONSULTA = v_sks(i);
            
        COMMIT;
        v_total := v_total + v_sks.COUNT;
        DBMS_OUTPUT.PUT_LINE('   ... Hechos actualizados: ' || TO_CHAR(v_total, '999,999,999'));
    END LOOP;
    CLOSE c_facts;
    DBMS_OUTPUT.PUT_LINE('>> Hechos poblados exitosamente: ' || v_total);
       EXCEPTION
       WHEN OTHERS THEN
           ROLLBACK;
           DBMS_OUTPUT.PUT_LINE('!! Error tras ' || v_total || ' filas: ' || SQLERRM);
           IF c_facts%ISOPEN THEN CLOSE c_facts; END IF;
           RAISE;	
END;
/
3. Trigger para Nuevos Registros (Garantiza que los datos futuros no queden vacíos)
Si en el futuro ingresan nuevas consultas desde la aplicación o scripts de prueba, este trigger asegura que siempre nazcan con su JSON estructurado:

sql
-- Ejecutar en red_clinica_user:
CREATE OR REPLACE TRIGGER TRG_CONSULTAS_BI_JSON
BEFORE INSERT ON CONSULTAS
FOR EACH ROW
BEGIN
    IF :NEW.EXPEDIENTE_CLINICO_JSON IS NULL THEN
        :NEW.EXPEDIENTE_CLINICO_JSON := JSON_OBJECT(
            'id_consulta'     VALUE :NEW.ID_CONSULTA,
            'motivo_consulta' VALUE :NEW.MOTIVO_CONSULTA,
            'prescripcion'    VALUE NVL(:NEW.PRESCRIPCION, 'Pendiente de evaluacion'),
            'honorarios'      VALUE :NEW.PRECIO_CONSULTA,
            'diagnosticos'    VALUE JSON_ARRAY(),
            'triaje_medico'   VALUE JSON_OBJECT(
                'prioridad'      VALUE 'NORMAL',
                'signos_vitales' VALUE JSON_OBJECT(
                    'presion_arterial' VALUE '120/80',
                    'temperatura_c'    VALUE 36.5,
                    'frecuencia_card'  VALUE 72
                )
            )
            RETURNING CLOB
        );
    END IF;
END;
/
4. Consulta de Auditoría (Para comprobar en clase o frente al profesor)
Ejecuta este SELECT para demostrar que los datos existentes se poblaron correctamente y se leen con sintaxis nativa JSON de Oracle:

sql
SELECT 
    c.ID_CONSULTA,
    c.MOTIVO_CONSULTA,
    -- Extracción directa con notación punto
    c.EXPEDIENTE_CLINICO_JSON.triaje_medico.prioridad AS PRIORIDAD,
    c.EXPEDIENTE_CLINICO_JSON.triaje_medico.signos_vitales.temperatura_c AS TEMP,
    -- Extracción con función estándar
    JSON_VALUE(c.EXPEDIENTE_CLINICO_JSON, '$.prescripcion') AS PRESCRIPCION
FROM CONSULTAS c
WHERE ROWNUM <= 5;
